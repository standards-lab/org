==================== a. the whole-forest baseline ====================
The volume-era shape: a projection whose base recurses from the root over every directory computing its path, joins the files, and takes the directory as a directive outside the derived table; the count twin and the page.
count: SELECT COUNT(*) FROM (WITH RECURSIVE tree (id, parent_id, path) AS (
    SELECT d.id, d.parent_id, CAST('' AS text)
    FROM blobfs_directory d
    WHERE d.parent_id IS NULL
  UNION ALL
    SELECT d.id, d.parent_id, t.path || '/' || d.name
    FROM blobfs_directory d
    JOIN tree t ON t.id = d.parent_id
)
SELECT f.id, f.directory_id, f.name, f.status, f.key, f.size, f.content_type, f.etag, f.version, f.created_at, f.updated_at, t.path || '/' || f.name AS path
FROM blobfs_file f
JOIN tree t ON t.id = f.directory_id) q WHERE q.directory_id = CAST($1 AS uuid)
